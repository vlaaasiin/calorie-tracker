        <input id="foodCalories" type="number" min="1" placeholder="ккал" class="w-24 bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-sm outline-none focus:border-emerald-500">

      </div>


      <div class="grid grid-cols-3 gap-2 mb-3">

        <input id="foodProtein" type="number" min="0" step="0.1" placeholder="Белки, г" class="bg-slate-50 border border-slate-200 rounded-xl px-2.5 py-2 text-xs outline-none focus:border-sky-500">

  <script>

    const tg = window.Telegram?.WebApp;

    if (tg) {

      tg.ready();

      tg.expand();

      if (tg.initDataUnsafe?.user?.first_name) {

        document.getElementById('userName').innerText = `Привет, ${tg.initDataUnsafe.user.first_name}!`;

      }

    }


    function getLocalDateString(dateObj = new Date()) {

      const year = dateObj.getFullYear();

      const month = String(dateObj.getMonth() + 1).padStart(2, '0');

      const day = String(dateObj.getDate()).padStart(2, '0');

      return `${year}-${month}-${day}`;

    }


    const todayStr = getLocalDateString();

    let selectedDate = todayStr;

    let dailyGoal = parseInt(localStorage.getItem('cal_daily_goal')) || 2000;


    // Записи храним в объекте по датам: { "2026-11-20": [ ... ] }

    let historyByDate = JSON.parse(localStorage.getItem('cal_history_by_date') || '{}');


    // Переносим старые записи из ключа cal_meals на сегодня, если они были

    const legacyMeals = JSON.parse(localStorage.getItem('cal_meals') || '[]');

    if (legacyMeals.length > 0 && !historyByDate[todayStr]) {

      historyByDate[todayStr] = legacyMeals.map(m => ({

        name: m.name,

        cals: m.cals,

        protein: 0,

        fat: 0,

        carbs: 0,

        time: m.time || '—'

      }));

      localStorage.setItem('cal_history_by_date', JSON.stringify(historyByDate));

      localStorage.removeItem('cal_meals');

    }


    document.getElementById('datePicker').value = selectedDate;

    document.getElementById('goalInput').value = dailyGoal;


    function toggleGoalEditor() {

      const editor = document.getElementById('goalEditor');

      editor.classList.toggle('hidden');

      document.getElementById('goalInput').value = dailyGoal;

    }


    function saveGoal() {

      const val = parseInt(document.getElementById('goalInput').value);

      if (isNaN(val) || val < 500 || val > 10000) return;

      dailyGoal = val;

      localStorage.setItem('cal_daily_goal', String(dailyGoal));

      document.getElementById('goalEditor').classList.add('hidden');

      if (tg?.HapticFeedback) tg.HapticFeedback.notificationOccurred('success');

      render();

    }


    function shiftDate(deltaDays) {

      const [y, m, d] = selectedDate.split('-').map(Number);

      const dt = new Date(y, m - 1, d + deltaDays);

      selectedDate = getLocalDateString(dt);

      document.getElementById('datePicker').value = selectedDate;

      render();

    }


    function onDatePick(val) {

      if (!val) return;

      selectedDate = val;

      render();

    }


    function formatNumber(num) {

      return Number.isInteger(num) ? String(num) : num.toFixed(1).replace(/\.0$/, '');

    }


    function render() {

      document.getElementById('goalBadge').innerText = dailyGoal;

      document.getElementById('todayTag').classList.toggle('hidden', selectedDate !== todayStr);


      const dayMeals = historyByDate[selectedDate] || [];


      const totalCals = dayMeals.reduce((sum, item) => sum + (Number(item.cals) || 0), 0);

      const totalProtein = dayMeals.reduce((sum, item) => sum + (Number(item.protein) || 0), 0);

      const totalFat = dayMeals.reduce((sum, item) => sum + (Number(item.fat) || 0), 0);

      const totalCarbs = dayMeals.reduce((sum, item) => sum + (Number(item.carbs) || 0), 0);


      document.getElementById('consumedDisplay').innerText = `${Math.round(totalCals)} / ${dailyGoal} ккал`;


      const remaining = dailyGoal - Math.round(totalCals);

      document.getElementById('remainingDisplay').innerText = remaining >= 0

        ? `Осталось: ${remaining} ккал`

        : `Превышение на: ${Math.abs(remaining)} ккал`;


      const percent = Math.min(100, Math.max(0, Math.round((totalCals / dailyGoal) * 100)));

      const bar = document.getElementById('progressBar');

      bar.style.width = `${percent}%`;

      bar.className = remaining >= 0

        ? 'bg-emerald-500 h-3 rounded-full transition-all duration-300'

        : 'bg-rose-500 h-3 rounded-full transition-all duration-300';


      document.getElementById('totalProtein').innerText = `${formatNumber(totalProtein)} г`;

      document.getElementById('totalFat').innerText = `${formatNumber(totalFat)} г`;

      document.getElementById('totalCarbs').innerText = `${formatNumber(totalCarbs)} г`;


      const listContainer = document.getElementById('mealsList');

      listContainer.innerHTML = '';


      if (dayMeals.length === 0) {

        listContainer.innerHTML = '<div class="text-center py-6 text-xs text-slate-400 bg-white rounded-2xl border border-slate-100">За этот день записей пока нет</div>';

        return;

      }


      dayMeals.forEach((meal, index) => {

        const item = document.createElement('div');

        item.className = 'flex justify-between items-center p-3 bg-white border border-slate-100 rounded-xl text-sm shadow-sm';

        item.innerHTML = `

          <div>

            <div class="font-medium">${meal.name}</div>

            <div class="text-xs text-slate-400">

              ${meal.time} · Б ${formatNumber(meal.protein || 0)} · Ж ${formatNumber(meal.fat || 0)} · У ${formatNumber(meal.carbs || 0)}

            </div>

          </div>

          <div class="flex items-center gap-3">

            <span class="font-bold text-slate-700 whitespace-nowrap">${Math.round(meal.cals)} ккал</span>

            <button onclick="removeEntry(${index})" class="text-slate-300 hover:text-rose-500 text-xs px-1">✕</button>

          </div>

        `;

        listContainer.appendChild(item);

      });

    }


    function addEntry() {

      const nameInput = document.getElementById('foodName');

      const calInput = document.getElementById('foodCalories');

      const pInput = document.getElementById('foodProtein');

      const fInput = document.getElementById('foodFat');

      const cInput = document.getElementById('foodCarbs');


      const name = nameInput.value.trim();

      const cals = parseFloat(calInput.value);

      const protein = parseFloat(pInput.value) || 0;

      const fat = parseFloat(fInput.value) || 0;

      const carbs = parseFloat(cInput.value) || 0;


      if (!name || isNaN(cals) || cals <= 0) return;


      const time = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      if (!historyByDate[selectedDate]) {

        historyByDate[selectedDate] = [];

      }


      historyByDate[selectedDate].unshift({ name, cals, protein, fat, carbs, time });

      localStorage.setItem('cal_history_by_date', JSON.stringify(historyByDate));


      nameInput.value = '';

      calInput.value = '';

      pInput.value = '';

      fInput.value = '';

      cInput.value = '';


      if (tg?.HapticFeedback) {

        tg.HapticFeedback.impactOccurred('light');

      }


      render();

    }


    function removeEntry(index) {

      if (!historyByDate[selectedDate]) return;

      historyByDate[selectedDate].splice(index, 1);

      localStorage.setItem('cal_history_by_date', JSON.stringify(historyByDate));

      if (tg?.HapticFeedback) {

        tg.HapticFeedback.notificationOccurred('warning');

      }

      render();

    }


    render();
